roads = int(input())

best_road = 0
best_height = -1

for road in range(1, roads + 1):
    tunnels = int(input())
    min_height = 10**9

    for _ in range(tunnels):
        h = int(input())
        if h < min_height:
            min_height = h

    if min_height > best_height:
        best_height = min_height
        best_road = road

print(best_road, best_height)
