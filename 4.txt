import math

k = float(input("Введите длину палки x = k: "))
h = float(input("Введите шаг h: "))
y = 2.0

print("y\tУгол альфа (в градусах)")
while y <= 3.0 + 1e-9: 
    if y <= k:
        
        alpha_rad = math.acos(y / k)
        alpha_deg = math.degrees(alpha_rad)
        print(f"{y:.2f}\t{alpha_deg:.2f}")
    else:
        print(f"{y:.2f}\tНевозможно (палка короче расстояния y)")
    y += h
