def is_prime(num):
    if num <= 1: return False
    if num == 2: return True
    if num % 2 == 0: return False
    for i in range(3, int(num**0.5) + 1, 2):
        if num % i == 0:
            return False
    return True
print("x\tЧисло\tПростое?")
for x in range(0, 29):
    value = 2 * (x**2) + 29
    if is_prime(value):
        print(f"{x}\t{value}\t\tДа")
