A = float(input("Введите стипендию A: "))
B = float(input("Введите превышение расходов B: "))
current_expenses = A + B
total_from_parents = 0.0
for month in range(1, 11): 
    shortage = current_expenses - A
    total_from_parents += shortage
    current_expenses *= 1.03
print(f"Нужно попросить у родителей: {total_from_parents:.2f} руб.")
